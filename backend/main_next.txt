
# PUBLIC VAPID KEY

@app.get("/api/vapid-public-key")
def get_vapid_public_key():

    if not VAPID_PUBLIC_KEY:
        raise HTTPException(
            status_code=500,
            detail="VAPID public key is missing.",
        )

    return {
        "publicKey": VAPID_PUBLIC_KEY,
    }


# SAVE PUSH SUBSCRIPTION

@app.post("/api/subscribe")
def subscribe(request: PushSubscriptionRequest):

    with SessionLocal() as session:

        existing = session.get(
            PushSubscriptionDB,
            request.endpoint
        )

        if existing:

            existing.p256dh = request.keys["p256dh"]
            existing.auth = request.keys["auth"]

        else:

            session.add(
                PushSubscriptionDB(
                    endpoint=request.endpoint,
                    p256dh=request.keys["p256dh"],
                    auth=request.keys["auth"],
                )
            )

        session.commit()

        count = session.scalar(
            select(func.count()).select_from(
                PushSubscriptionDB
            )
        )

    return {
        "message": "Push subscription saved!",
        "total": count,
    }


# TEST PUSH NOTIFICATION

@app.post("/api/test-push")
def test_push(request: TestPushRequest):

    with SessionLocal() as session:

        subscription = session.get(
            PushSubscriptionDB,
            request.endpoint
        )

        if not subscription:
            raise HTTPException(
                status_code=404,
                detail="Push subscription not found.",
            )

        try:

            send_push(
                subscription,
                {
                    "title": "RemindMe Test Notification",
                    "body": "PostgreSQL-backed push notifications are working!",
                    "tag": "remindme-test",
                    "url": "/",
                }
            )

            return {
                "message": "Test push notification sent successfully!"
            }

        except WebPushException as error:

            response = getattr(error, "response", None)

            status = (
                response.status_code
                if response is not None
                else None
            )

            logger.exception("Test push failed.")

            if status in (404, 410):
                session.delete(subscription)
                session.commit()

            raise HTTPException(
                status_code=502,
                detail=f"Push delivery failed: HTTP {status}",
            )
